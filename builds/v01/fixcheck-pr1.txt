"""Hard per-run budgets, 429 backoff, and checkpointed runs that resume.

A run appends one JSON line per finished item to its results file. Re-running
with the same file skips finished items and carries their spend forward, so the
budget covers the whole run across resumes. When the budget stops a run, the
outcome is marked partial and still holds every finished record.
"""

from __future__ import annotations

import json
import os
import time
from collections.abc import Callable, Sequence
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

from ledgerline.llm.errors import BudgetExceededError, RateLimitedError


class BudgetGuard:
    def __init__(self, limit_usd: float, spent_usd: float = 0.0) -> None:
        if limit_usd < 0:
            raise ValueError("budget must be >= 0")
        self.limit_usd = limit_usd
        self.spent_usd = spent_usd

    @property
    def remaining_usd(self) -> float:
        return self.limit_usd - self.spent_usd

    def precheck(self, worst_case_usd: float) -> None:
        """Refuse a call whose worst-case cost could cross the limit."""
        if self.spent_usd + worst_case_usd > self.limit_usd:
            raise BudgetExceededError(self.spent_usd, worst_case_usd, self.limit_usd)

    def charge(self, cost_usd: float) -> None:
        self.spent_usd += cost_usd


def is_rate_limit(exc: BaseException) -> bool:
    if getattr(exc, "status_code", None) == 429:
        return True
    response = getattr(exc, "response", None)
    return getattr(response, "status_code", None) == 429


def with_backoff[T](
    call: Callable[[], T],
    *,
    retries: int = 5,
    base_delay_s: float = 1.0,
    max_delay_s: float = 30.0,
    sleep: Callable[[float], None] = time.sleep,
) -> T:
    """Run `call`, retrying only HTTP 429s with exponential backoff."""
    for attempt in range(retries + 1):
        try:
            return call()
        except Exception as exc:
            if not is_rate_limit(exc):
                raise
            if attempt == retries:
                raise RateLimitedError(f"still rate limited after {retries} retries") from exc
            sleep(min(max_delay_s, base_delay_s * 2**attempt))
    raise AssertionError("unreachable")


class Checkpoint:
    """Append-only JSONL. Each line is one of:

    - a finished item: `{"key", "cost_usd", "record"}`
    - spend from an unfinished item (budget stop or crash): `{"key", "cost_usd",
      "record": null}`. It is not "done", but its cost still counts on resume.
    """

    def __init__(self, path: Path | str) -> None:
        self.path = Path(path)

    def load(self) -> tuple[dict[str, dict[str, Any]], float]:
        """Return (finished entries by key, total spend recorded so far).

        A torn final line (a run killed mid-write: no trailing newline, not valid JSON)
        is cut off the file, so later appends start on a clean line. Any other line that
        is not valid JSON is skipped, never treated as the end of the file."""
        done: dict[str, dict[str, Any]] = {}
        spent = 0.0
        if not self.path.exists():
            return done, spent
        self._repair_torn_tail()
        with self.path.open(encoding="utf-8") as fh:
            for line in fh:
                if not line.strip():
                    continue
                try:
                    entry = json.loads(line)
                except json.JSONDecodeError:
                    continue
                spent += float(entry["cost_usd"])
                if entry["record"] is not None:
                    done[entry["key"]] = entry
        return done, spent

    def _repair_torn_tail(self) -> None:
        data = self.path.read_bytes()
        if not data or data.endswith(b"\n"):
            return
        start = data.rfind(b"\n") + 1
        try:
            json.loads(data[start:])
        except json.JSONDecodeError:
            with self.path.open("r+b") as fh:
                fh.truncate(start)
            return
        with self.path.open("ab") as fh:  # complete line, missing only its newline
            fh.write(b"\n")

    def append(self, key: str, cost_usd: float, record: dict[str, Any] | None) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        line = json.dumps({"key": key, "cost_usd": cost_usd, "record": record}, sort_keys=True)
        with self.path.open("a+b") as fh:
            # Never glue a record onto a torn line: start on a fresh line if needed.
            fh.seek(0, os.SEEK_END)

            fh.write(prefix + line.encode("utf-8") + b"\n")
            fh.flush()
            os.fsync(fh.fileno())

def run_checkpointed[T](
    items: Sequence[T],
    key: Callable[[T], str],
    work: Callable[[T, BudgetGuard], ItemResult],
    checkpoint: Checkpoint,
    budget_usd: float,
) -> RunOutcome:
    """Run `work` over `items`, resuming from `checkpoint` and stopping at the budget.

    `work` must pre-check and charge the guard for each call it makes (the gateway
    does). The budget is the run's total: spend already in the checkpoint counts.
    """
    done, already_spent = checkpoint.load()
    guard = BudgetGuard(budget_usd, spent_usd=already_spent)
    outcome = RunOutcome(total_items=len(items))
    for item in items:
        item_key = key(item)
        if item_key in done:
            outcome.records.append(done[item_key]["record"])
            outcome.resumed += 1
            continue
        spent_before = guard.spent_usd
        try:
            result = work(item, guard)
        except BudgetExceededError as exc:
            _record_unfinished_spend(checkpoint, item_key, guard.spent_usd - spent_before)
            outcome.aborted = True
            outcome.abort_reason = str(exc)
            break
        except BaseException:
            _record_unfinished_spend(checkpoint, item_key, guard.spent_usd - spent_before)
            raise
        # Count the larger of what the guard saw and what `work` reports, so spend
        # that bypassed the guard still reaches the budget and the checkpoint.
        seen = guard.spent_usd - spent_before
        if result.cost_usd > seen:
            guard.charge(result.cost_usd - seen)
        cost = guard.spent_usd - spent_before
        checkpoint.append(item_key, cost, result.record)
        outcome.records.append(result.record)
        outcome.newly_completed += 1
    outcome.spent_usd = guard.spent_usd
    return outcome


def _record_unfinished_spend(checkpoint: Checkpoint, key: str, spent: float) -> None:
    if spent > 0:
        checkpoint.append(key, spent, None)

class _Grain(FunctionElement[Any]):
    type = Date()
    inherit_cache = True
    grain = ""


class GrainDay(_Grain):
    name = "grain_day"
    grain = "day"
    inherit_cache = True


class GrainIsoWeek(_Grain):
    name = "grain_iso_week"
    grain = "iso_week"
    inherit_cache = True


class GrainMonth(_Grain):
    name = "grain_month"
    grain = "month"
    inherit_cache = True


class GrainQuarter(_Grain):
    name = "grain_quarter"
    grain = "quarter"
    inherit_cache = True


class GrainYear(_Grain):
    name = "grain_year"
    grain = "year"
    inherit_cache = True


GRAINS: dict[str, type[_Grain]] = {
    "day": GrainDay,
    "iso_week": GrainIsoWeek,
    "month": GrainMonth,
    "quarter": GrainQuarter,
    "year": GrainYear,
}

# Postgres: date_trunc's 'week' is the ISO week (Monday start).
_PG_UNITS = {"iso_week": "week", "month": "month", "quarter": "quarter", "year": "year"}


@compiles(_Grain, "postgresql")
def _grain_pg(element: _Grain, compiler: SQLCompiler, **kw: Any) -> str:
    expr = compiler.process(element.clauses, **kw)
    if element.grain == "day":
        return f"CAST({expr} AS DATE)"
    return f"CAST(date_trunc('{_PG_UNITS[element.grain]}', {expr}) AS DATE)"


@compiles(_Grain, "sqlite")
def _grain_sqlite(element: _Grain, compiler: SQLCompiler, **kw: Any) -> str:
    expr = compiler.process(element.clauses, **kw)
    if element.grain == "day":
        return f"date({expr})"
    if element.grain == "iso_week":
        # 'weekday 0' moves forward to Sunday (or stays); 6 days back is that ISO Monday.
        return f"date({expr}, 'weekday 0', '-6 days')"
    if element.grain == "month":
        return f"date({expr}, 'start of month')"
    if element.grain == "quarter":
        back = f"((CAST(strftime('%m', {expr}) AS INTEGER) - 1) % 3)"
        return f"date({expr}, 'start of month', '-' || {back} || ' months')"
    return f"date({expr}, 'start of year')"


@compiles(_Grain)
def _grain_other(element: _Grain, compiler: SQLCompiler, **kw: Any) -> str:
    raise CompileError(f"calendar grains are not implemented for {compiler.dialect.name}")


class AsDatetime(FunctionElement[Any]):
    """An operand normalised to a datetime: CAST AS TIMESTAMP on Postgres; on SQLite,
    strftime to one text format (millisecond precision), whatever format the stored
    DATE or DATETIME text uses."""

    type = DateTime()
    name = "as_datetime"
    inherit_cache = True


@compiles(AsDatetime, "postgresql")
def _as_datetime_pg(element: AsDatetime, compiler: SQLCompiler, **kw: Any) -> str:
    return f"CAST({compiler.process(element.clauses, **kw)} AS TIMESTAMP)"


@compiles(AsDatetime, "sqlite")
def _as_datetime_sqlite(element: AsDatetime, compiler: SQLCompiler, **kw: Any) -> str:
    return f"strftime('%Y-%m-%d %H:%M:%f', {compiler.process(element.clauses, **kw)})"


@compiles(AsDatetime)

class _Tables:
    def __init__(self, plan: QueryPlan) -> None:
        columns: dict[str, dict[str, TypeEngine[Any]]] = {plan.base_alias: {}}
        names: dict[str, tuple[str, str | None]] = {
            plan.base_alias: (plan.base_table, plan.base_schema)
        }
        for join in plan.joins:
            names[join.alias] = (join.table, join.db_schema)
            columns.setdefault(join.parent_alias, {}).setdefault(join.parent_column, NullType())
            columns.setdefault(join.alias, {}).setdefault(join.column, NullType())
        for rfield in _fields(plan):
            columns.setdefault(rfield.alias, {})[rfield.column] = _TYPES[rfield.type]
        self.tables: dict[str, FromClause] = {}
        for alias, (name, schema) in names.items():
            cols = [column(c, t) for c, t in sorted(columns.get(alias, {}).items())]
            self.tables[alias] = table(name, *cols, schema=schema).alias(alias)

    def col(self, rfield: ResolvedField) -> ColumnElement[Any]:
        return self.tables[rfield.alias].c[rfield.column]


def _fields(plan: QueryPlan) -> list[ResolvedField]:
    out: list[ResolvedField] = list(plan.select)
    out += [m.field for m in plan.metrics if m.field is not None]
    out += [g.field for g in plan.group_by]
    out += [o.field for o in plan.order_by if o.field is not None]
    for item in plan.where:
        predicates = item.predicates if isinstance(item, PredicateGroup) else (item,)
        for predicate in predicates:
            out.append(predicate.field)
            if predicate.other is not None:
                out.append(predicate.other)
    return out


def _bind(value: Any, col: ColumnElement[Any]) -> ColumnElement[Any]:
    """A bound parameter typed like its column. Passing a bare Python value lets
    SQLAlchemy render some values (booleans) as SQL literals; this never does."""
    return bindparam(None, value, type_=col.type)


def _comparison_operands(
    tables: _Tables, predicate: Predicate
) -> tuple[ColumnElement[Any], ColumnElement[Any]]:
    col = tables.col(predicate.field)
    assert predicate.other is not None
    other = tables.col(predicate.other)
    if {predicate.field.type, predicate.other.type} == {"date", "datetime"}:
        # A DATE against a DATETIME: compare both as datetimes (a DATE is its midnight).
        return AsDatetime(col), AsDatetime(other)
    return col, other


def _predicate(tables: _Tables, predicate: Predicate) -> ColumnElement[bool]:
    op = predicate.op
    if predicate.other is not None:
        col, rhs = _comparison_operands(tables, predicate)
    else:
        col = tables.col(predicate.field)
        single = op in ("eq", "neq", "gt", "gte", "lt", "lte")
        rhs = _bind(predicate.values[0], col) if single else None
    if op == "eq":
        return col == rhs
    if op == "neq":
        return col != rhs
    if op == "gt":
        return col > rhs
    if op == "gte":
        return col >= rhs
    if op == "lt":
        return col < rhs
    if op == "lte":
        return col <= rhs
    if op == "between":
        return col.between(_bind(predicate.values[0], col), _bind(predicate.values[1], col))
    if op == "in":
        return col.in_([_bind(v, col) for v in predicate.values])
    if op == "is_null":
        return col.is_(None) if predicate.values[0] else col.is_not(None)
    if op == "ilike_contains":
        return col.icontains(predicate.values[0], autoescape=True)
    raise ValueError(f"unknown operator {op!r}")


def _metric(tables: _Tables, metric: PlanMetric) -> ColumnElement[Any]:
    if metric.field is None:
        return func.count()
    col = tables.col(metric.field)
    arg = col.distinct() if metric.distinct else col
    if metric.agg in ("count", "count_distinct"):
        return func.count(arg)
    return getattr(func, metric.agg)(arg)


def _compare(expr: ColumnElement[Any], op: str, values: tuple[float, ...]) -> ColumnElement[bool]:
    if op == "between":
        return expr.between(_bind(values[0], expr), _bind(values[1], expr))
    value = _bind(values[0], expr)
    comparisons = {
        "eq": expr == value,
        "neq": expr != value,
        "gt": expr > value,
        "gte": expr >= value,
        "lt": expr < value,
        "lte": expr <= value,
    }
    return comparisons[op]


def _label(labelled: dict[str, ColumnElement[Any]], label: str, expr: ColumnElement[Any]) -> None:
    if label in labelled:
        # The validator rejects this (alias_collision); never let one label shadow another.
        raise ValueError(f"duplicate output label {label!r}")
    labelled[label] = expr


def build(plan: QueryPlan) -> Select[Any]:
    tables = _Tables(plan)
    source = tables.tables[plan.base_alias]
    for join in plan.joins:
        parent = tables.tables[join.parent_alias]
        child = tables.tables[join.alias]
        source = source.join(child, parent.c[join.parent_column] == child.c[join.column])

    labelled: dict[str, ColumnElement[Any]] = {}
    group_exprs: list[ColumnElement[Any]] = []
    columns: list[ColumnElement[Any]] = []
    if plan.intent_type == "row_list":
        columns = [tables.col(f).label(f.label) for f in plan.select]
    else:
        for group in plan.group_by:
            expr = tables.col(group.field)
            if group.grain is not None:
                expr = GRAINS[group.grain](expr)
            group_exprs.append(expr)
            columns.append(expr.label(group.label))
            _label(labelled, group.label, expr)
        for metric in plan.metrics:
            expr = _metric(tables, metric)
            columns.append(expr.label(metric.label))
            _label(labelled, metric.label, expr)

    stmt = select(*columns).select_from(source)
    conditions: list[ColumnElement[bool]] = []
    for item in plan.where:
        if isinstance(item, PredicateGroup):
            parts = [_predicate(tables, p) for p in item.predicates]
            conditions.append(or_(*parts) if item.mode == "any_of" else and_(*parts))
        else:
            conditions.append(_predicate(tables, item))
    if conditions:
        stmt = stmt.where(and_(*conditions))
    if group_exprs:
        stmt = stmt.group_by(*group_exprs)
    if plan.having:
        stmt = stmt.having(
            and_(*[_compare(labelled[h.metric], h.op, h.values) for h in plan.having])
        )
    order: list[ColumnElement[Any]] = []
    for item in plan.order_by:
        expr = tables.col(item.field) if item.field is not None else labelled[item.label or ""]
        order.append(expr.desc() if item.direction == "desc" else expr.asc())
    if order:
        stmt = stmt.order_by(*order)
    if plan.limit is not None:
        stmt = stmt.limit(plan.limit)
    return stmt


def dialect_for(name: str) -> Dialect:
    if name == "postgresql":
        return postgresql.psycopg.dialect()  # type: ignore[no-any-return]
    if name == "sqlite":
        return sqlite.pysqlite.dialect()  # type: ignore[no-any-return]
    raise ValueError(f"unsupported dialect {name!r}")


def compile_sql(stmt: Select[Any], dialect_name: str) -> tuple[str, dict[str, Any]]:

    # values

    def convert(self, value: Scalar | None, ftype: FieldType, location: str) -> Value | None:
        ok = True
        out: Value | None = None
        if ftype == "integer":
            ok = isinstance(value, int) and not isinstance(value, bool)
            out = value
        elif ftype == "number":
            ok = isinstance(value, int | float) and not isinstance(value, bool)
            out = value
        elif ftype == "text":
            ok = isinstance(value, str)
            out = value
        elif ftype == "boolean":
            ok = isinstance(value, bool)
            out = value
        elif ftype == "date":
            try:
                out = date.fromisoformat(value) if isinstance(value, str) else None
            except ValueError:
                out = None
            ok = out is not None
        else:
            try:
                out = naive_utc(datetime.fromisoformat(value)) if isinstance(value, str) else None
            except ValueError:
                out = None
            ok = out is not None
        if not ok:
            self.fail("value_type", location, f"{value!r} is not a valid {ftype}")
            return None
        return out

    def categorical(self, value: Value, fdef: FieldDef, location: str) -> Value | None:
        if not (fdef.categorical and fdef.allowed_values and isinstance(value, str)):
            return value
        lowered = value.lower()
        for allowed in fdef.allowed_values:
            if allowed.lower() == lowered:
                return allowed
        for alias, target in fdef.value_aliases.items():
            if alias.lower() == lowered:
                return target
        self.fail(
            "value_not_allowed",
            location,
            f"{value!r} is not one of {fdef.column}'s allowed values",
        )
        return None

    def check_shape(self) -> bool:
        intent = self.intent
        loc = "intent_type"
        if intent.intent_type == "unanswerable":
            populated = (
                intent.select
                or intent.metrics
                or intent.filters
                or intent.having
                or intent.group_by
                or intent.date_range
                or intent.order_by
                or intent.limit is not None
            )
            if not intent.unanswerable_reason:
                self.fail("intent_shape", loc, "unanswerable needs an unanswerable_reason")
            if populated:
                self.fail("intent_shape", loc, "unanswerable takes no query fields")
            return False
        if intent.unanswerable_reason is not None:
            self.fail("intent_shape", loc, "only unanswerable takes an unanswerable_reason")
        if intent.entity is None:
            self.fail("intent_shape", "entity", f"{intent.intent_type} needs an entity")
        if intent.intent_type == "row_list":
            if not intent.select:
                self.fail("intent_shape", loc, "row_list needs select")
            if intent.metrics or intent.group_by or intent.having:
                self.fail("intent_shape", loc, "row_list takes no metrics, group_by or having")
        else:
            if not intent.metrics:
                self.fail("intent_shape", loc, f"{intent.intent_type} needs metrics")
            if intent.select:
                self.fail("intent_shape", loc, f"{intent.intent_type} takes no select")
            if intent.intent_type == "summary" and intent.group_by:
                self.fail("intent_shape", loc, "summary takes no group_by; use grouped_summary")
            if intent.intent_type == "grouped_summary" and not intent.group_by:
                self.fail("intent_shape", loc, "grouped_summary needs group_by")
        if intent.having and not intent.group_by:
            self.fail("having_without_group_by", "having", "having needs a group_by")
        return intent.entity is not None

    def check_condition(self, cond: Condition, loc: str) -> Predicate | None:
        resolved = self.resolve(cond.field, f"{loc}.field")
        if resolved is None:
            return None
        rfield, fdef = resolved
        if not fdef.filterable:
            self.fail("not_filterable", f"{loc}.field", f"{cond.field} is not filterable")
            return None
        op = cond.op
        if op in _RANGE_OPS and rfield.type not in _ORDERABLE:
            self.fail("operator_type", f"{loc}.op", f"{op} does not apply to {rfield.type}")
            return None
        if op == "ilike_contains" and rfield.type != "text":
            self.fail("operator_type", f"{loc}.op", "ilike_contains needs a text field")
            return None

        if cond.other_field is not None:
            return self.check_comparison(cond, rfield, loc)

        if op == "in":
            if cond.values is None or cond.value is not None:
                self.fail("value_shape", loc, "in takes `values`")
                return None
            if not cond.values:
                self.fail("empty_in", f"{loc}.values", "in needs at least one value")
                return None
            return self._values_predicate(rfield, fdef, op, cond.values, loc)
        if op == "between":
            if cond.values is None or len(cond.values) != 2 or cond.value is not None:
                self.fail("value_shape", loc, "between takes `values: [low, high]`")
                return None
            predicate = self._values_predicate(rfield, fdef, op, cond.values, loc)
            if predicate is None:
                return None
            low, high = predicate.values
            if low > high:  # type: ignore[operator]
                self.fail("reversed_between", f"{loc}.values", f"{low!r} > {high!r}")
                return None
            return predicate
        if op == "is_null":
            if cond.values is not None or not isinstance(cond.value, bool | None):
                self.fail("value_shape", loc, "is_null takes `value: true|false` or nothing")
                return None
            return Predicate(rfield, op, (cond.value is not False,))
        # eq, neq, gt, gte, lt, lte, ilike_contains
        if cond.values is not None or cond.value is None:
            self.fail("value_shape", loc, f"{op} takes one `value`")
            return None
        if op == "ilike_contains":
            if not isinstance(cond.value, str) or not cond.value:
                self.fail("value_type", f"{loc}.value", "ilike_contains needs non-empty text")
                return None
            return Predicate(rfield, op, (cond.value,))
        return self._values_predicate(rfield, fdef, op, [cond.value], loc)

    def _values_predicate(
        self, rfield: ResolvedField, fdef: FieldDef, op: str, raw: list[Scalar], loc: str
    ) -> Predicate | None:
        values: list[Value] = []
        for index, item in enumerate(raw):
            converted = self.convert(item, rfield.type, f"{loc}.values[{index}]")
            if converted is None:
                return None
            if op in ("eq", "neq", "in"):
                converted = self.categorical(converted, fdef, f"{loc}.values[{index}]")
                if converted is None:
                    return None
            values.append(converted)
        return Predicate(rfield, op, tuple(values))

    def check_comparison(
        self, cond: Condition, rfield: ResolvedField, loc: str
    ) -> Predicate | None:
        assert cond.other_field is not None
        if cond.op not in COMPARISON_OPS or cond.value is not None or cond.values is not None:
            self.fail("value_shape", loc, "other_field works with eq..lte and no value")
            return None
        resolved = self.resolve(cond.other_field, f"{loc}.other_field")
        if resolved is None:
            return None
        other, other_def = resolved
        if not other_def.filterable:
            self.fail("not_filterable", f"{loc}.other_field", f"{cond.other_field} not filterable")
            return None
        if _family(rfield.type) != _family(other.type):
            self.fail(
                "column_comparison_type",
                loc,
                f"cannot compare {rfield.type} with {other.type}",
            )
            return None
        return Predicate(rfield, cond.op, (), other)

    def check_filters(self) -> list[Predicate | PredicateGroup]:
        out: list[Predicate | PredicateGroup] = []
        for index, item in enumerate(self.intent.filters):
            loc = f"filters[{index}]"
            if isinstance(item, Condition):
                predicate = self.check_condition(item, loc)
                if predicate is not None:
                    out.append(predicate)
                continue
            out_group = self.check_group(item, loc)
            if out_group is not None:
                out.append(out_group)
        return out

    def check_metrics(self, group_labels: set[str]) -> list[PlanMetric]:
        """Metric labels must be unique and must not reuse a group_by label: a metric
        labelled like a group would shadow it in HAVING and ORDER BY."""
        out: list[PlanMetric] = []
        labels: set[str] = set()
        for index, metric in enumerate(self.intent.metrics):
            loc = f"metrics[{index}]"
            rfield: ResolvedField | None = None
            if metric.field is None:
                if metric.agg != "count":
                    self.fail("value_shape", loc, f"{metric.agg} needs a field")
                    continue
            else:
                resolved = self.resolve(metric.field, f"{loc}.field")
                if resolved is None:
                    continue
                rfield, fdef = resolved
                if metric.agg in ("sum", "avg") and rfield.type not in NUMERIC_TYPES:
                    self.fail("aggregate_type", loc, f"{metric.agg} needs a numeric field")
                    continue
                if metric.agg in ("min", "max") and rfield.type not in _ORDERABLE:
                    self.fail("aggregate_type", loc, f"{metric.agg} needs a number or date")
                    continue
                if metric.agg not in ("count", "count_distinct") and not fdef.aggregatable:
                    self.fail("not_aggregatable", loc, f"{metric.field} is not aggregatable")
                    continue
            if metric.alias is not None:
                if not _ALIAS.match(metric.alias):
                    self.fail("invalid_alias", f"{loc}.alias", f"{metric.alias!r} is not a slug")
                    continue
                label = metric.alias
                if label in group_labels:
                    self.fail(
                        "alias_collision",
                        f"{loc}.alias",
                        f"{label!r} is already a group_by label",
                    )
                    continue
            else:
                label = metric.agg if rfield is None else f"{metric.agg}_{rfield.name}"
                suffix = 2
                base_label = label
                while label in labels or label in group_labels:
                    label = f"{base_label}_{suffix}"
                    suffix += 1
            if label in labels:
                self.fail("duplicate_alias", f"{loc}.alias", f"{label!r} is used twice")
                continue
            labels.add(label)
            distinct = metric.distinct or metric.agg == "count_distinct"
            out.append(PlanMetric(label, metric.agg, rfield, distinct))
        return out

    def check_group_by(self) -> list[PlanGroup]:
        out: list[PlanGroup] = []
        for index, group in enumerate(self.intent.group_by):
            loc = f"group_by[{index}]"
            resolved = self.resolve(group.field, f"{loc}.field")
            if resolved is None:
                continue
            rfield, _ = resolved
            if group.grain is not None and rfield.type not in TEMPORAL_TYPES:
                self.fail("grain_type", loc, f"{group.grain} needs a date field")
                continue
            label = rfield.label if group.grain is None else f"{rfield.label}_{group.grain}"
            if label in {g.label for g in out}:
                self.fail("alias_collision", loc, f"{group.field} is grouped twice")
                continue
            out.append(PlanGroup(label, rfield, group.grain))
        return out

    def check_having(self, metrics: list[PlanMetric]) -> list[PlanHaving]:
        out: list[PlanHaving] = []
        labels = {metric.label for metric in metrics}
        for index, having in enumerate(self.intent.having):
            loc = f"having[{index}]"
            if having.metric not in labels:
                self.fail("unknown_metric", loc, f"no metric labelled {having.metric!r}")
                continue
            if having.op == "between":
                if having.values is None or len(having.values) != 2 or having.value is not None:
                    self.fail("value_shape", loc, "between takes `values: [low, high]`")
                    continue
                if having.values[0] > having.values[1]:
                    self.fail("reversed_between", loc, "having between is reversed")
                    continue
                out.append(PlanHaving(having.metric, "between", tuple(having.values)))
                continue
            if having.value is None or having.values is not None:
                self.fail("value_shape", loc, f"{having.op} takes one `value`")
                continue
            out.append(PlanHaving(having.metric, having.op, (having.value,)))
        return out

    def check_order(self, metrics: list[PlanMetric], groups: list[PlanGroup]) -> list[PlanOrder]:
        out: list[PlanOrder] = []
        metric_labels = {metric.label for metric in metrics}
        for index, order in enumerate(self.intent.order_by):
            loc = f"order_by[{index}]"
            if self.intent.intent_type == "row_list":
                resolved = self.resolve(order.target, f"{loc}.target")
                if resolved is not None:
                    out.append(PlanOrder(None, resolved[0], order.direction))
                continue
            if order.target in metric_labels:
                out.append(PlanOrder(order.target, None, order.direction))
                continue
            match = [g for g in groups if order.target in (g.label, _ref(g.field))]
            if not match:
                self.fail(
                    "invalid_order_by",
                    loc,
                    f"{order.target!r} is neither a metric alias nor a group_by field",
                )
                continue
            out.append(PlanOrder(match[0].label, None, order.direction))
        return out

    def check_limit(self) -> int | None:
        limit = self.intent.limit
        if limit is not None and not 0 < limit <= self.row_cap:
            self.fail("limit_out_of_range", "limit", f"limit must be 1..{self.row_cap}")
            return None
        return limit

    def run(self, allow_draft: bool) -> ValidationResult:
        if self.registry.draft and not allow_draft:
            self.fail("draft_registry", "registry", "the registry is a draft; pass allow_draft")
            return ValidationResult(self.issues)
        if not self.check_shape():
            if self.intent.intent_type == "unanswerable" and not self.issues:
                return ValidationResult(refusal_reason=self.intent.unanswerable_reason)
            return ValidationResult(self.issues)
        assert self.intent.entity is not None
        base = self._find_entity(self.intent.entity)
        if base is None:
            self.fail("unknown_entity", "entity", f"no entity {self.intent.entity!r}")
            return ValidationResult(self.issues)
        self.base = base
        self.base_def = self.registry.entities[base]
        self.paths = _shortest_paths(self.registry, base)

        select: list[ResolvedField] = []
        for index, ref in enumerate(self.intent.select):
            resolved = self.resolve(ref, f"select[{index}]")
            if resolved is not None:
                select.append(resolved[0])
        groups = self.check_group_by()
        metrics = self.check_metrics({g.label for g in groups})
        where = self.check_filters()
        where_dates = self.check_date_range(where)
        having = self.check_having(metrics)
        order = self.check_order(metrics, groups)
        limit = self.check_limit()
        if self.issues:
            return ValidationResult(self.issues)
        plan = QueryPlan(
            intent_type=self.intent.intent_type,
            base_entity=base,
            base_table=self.base_def.table,
            base_schema=self.base_def.db_schema,
            base_alias="t0",
            joins=tuple(self.joins),
            select=tuple(select),
            metrics=tuple(metrics),
            where=(*where, *where_dates),
            group_by=tuple(groups),
            having=tuple(having),
            order_by=tuple(order),
            limit=limit,
        )
        return ValidationResult(plan=plan)


def _family(ftype: str) -> str:
    if ftype in NUMERIC_TYPES:
        return "number"
    if ftype in TEMPORAL_TYPES:
        return "time"
    return ftype


def _ref(rfield: ResolvedField) -> str:
    return f"{rfield.entity}.{rfield.name}"


def _flatten(where: list[Predicate | PredicateGroup]) -> list[Predicate]:
    out: list[Predicate] = []
    for item in where:
        if isinstance(item, Predicate):
            out.append(item)
        else:
            out.extend(item.predicates)
    return out

"""The model gateway: one structured-output call, recorded, replayed, costed, budgeted.

Modes:
- `live`: call OpenRouter; nothing is stored.
- `record`: call OpenRouter and append the response to the replay store.
- `replay`: answer only from the replay store. A miss raises `ReplayMissError`;
  replay never falls back to a live call.

class Transport(Protocol):
    def __call__(
        self,
        *,
        model: str,
        messages: list[ChatMessage],
        schema_name: str,
        schema: dict[str, Any],
        max_tokens: int,
    ) -> RawResponse: ...


PriceLookup = Callable[[str], ModelPrice | None]

_FENCE = re.compile(r"^\s*```(?:json)?\s*(.*?)\s*```\s*$", re.DOTALL)


def parse_output[T: BaseModel](raw: RawResponse, schema: type[T]) -> T:
    """Validate the provider's parse, or fall back to parsing the text with Pydantic."""
    if raw.parsed is not None:
        try:
            return schema.model_validate(raw.parsed)
        except ValidationError as exc:
            raise ParseError(f"structured output failed validation: {exc}", raw.content) from exc
    text = raw.content.strip()
    fenced = _FENCE.match(text)
    if fenced:
        text = fenced.group(1)
    try:
        return schema.model_validate_json(text)
    except ValidationError as exc:
        raise ParseError(f"text did not validate: {exc.errors()[0]['msg']}", raw.content) from exc


class Gateway:
    def __init__(
        self,
        *,
        mode: LLMMode,
        store: ReplayStore | None,
        transport: Transport | None,
        price_lookup: PriceLookup,
        guard: BudgetGuard | None,
        max_tokens: int,
        backoff_sleep: Callable[[float], None] | None = None,
    ) -> None:
        if mode in ("record", "replay") and store is None:
            raise ValueError(f"{mode} mode needs a replay store")
        if mode in ("record", "live") and (transport is None or guard is None):
            raise ValueError(f"{mode} mode needs a transport and a budget guard")
        self.mode = mode
        self.store = store
        self.transport = transport
        self.price_lookup = price_lookup
        self.guard = guard
        self.max_tokens = max_tokens
        self._backoff_sleep = backoff_sleep

    def complete[T: BaseModel](
        self,
        *,
        model: str,
        prompt_version: str,
        messages: list[ChatMessage],
        schema: type[T],
    ) -> LLMResult[T]:
        json_schema = schema.model_json_schema()
        key = replay_key(model, prompt_version, messages, json_schema)

        if self.mode == "replay":
            assert self.store is not None
            record = self.store.get(key)
            raw = RawResponse.model_validate(record["raw"])
            return LLMResult[T](
                value=parse_output(raw, schema),
                raw=raw,
                cost_usd=float(record["cost_usd"]),
                cost_source="replay",
                replayed=True,
                key=key,
            )

        assert self.transport is not None
        assert self.guard is not None
        transport = self.transport
        price = self.price_lookup(model)
        if price is None:
            raise PricingUnavailableError(
                f"no price for {model!r}: the worst-case budget pre-check cannot run"
            )
        prompt_chars = len(canonical_json([m.model_dump() for m in messages])) + len(
            canonical_json(json_schema)
        )
        self.guard.precheck(worst_case_cost(price, prompt_chars, self.max_tokens))

        def call() -> RawResponse:
            return transport(
                model=model,
                messages=messages,
                schema_name=schema.__name__,
                schema=json_schema,
                max_tokens=self.max_tokens,
            )

        if self._backoff_sleep is None:
            raw = with_backoff(call)
        else:
            raw = with_backoff(call, sleep=self._backoff_sleep)
        cost, source = cost_from_usage(raw.usage, price)
        self.guard.charge(cost)
        if self.mode == "record":
            assert self.store is not None
            # Stored before parsing, so a replay reproduces a ParseError too.
            self.store.put(key, {"raw": raw.model_dump(), "cost_usd": cost, "model": model})
        return LLMResult[T](
            value=parse_output(raw, schema),
            raw=raw,
            cost_usd=cost,
            cost_source=source,
            replayed=False,
            key=key,
        )

        self,

        strict: bool = False,

            raise ValueError("OPENROUTER_API_KEY is not set; use replay mode without a key")
        self._settings = settings
        self._providers = providers
        self._strict = strict
        self._http_client = http_client

    def __call__(
        self,
        *,
        model: str,
        messages: list[ChatMessage],
        schema_name: str,
        schema: dict[str, Any],
        max_tokens: int,
    ) -> RawResponse:
        from langchain_openai import ChatOpenAI

        provider: dict[str, Any] = {"require_parameters": True}
        if self._providers:
            provider["order"] = self._providers
            provider["allow_fallbacks"] = False
        key = self._settings.openrouter_api_key
        assert key is not None
        llm = ChatOpenAI(
            model=model,
            base_url=self._settings.openrouter_base_url,
            api_key=key,
            temperature=0,
            max_retries=0,
            http_client=self._http_client,
            extra_body={
                "provider": provider,
                "usage": {"include": True},
                "max_tokens": max_tokens,
            },
        )
        structured = llm.with_structured_output(
            {**schema, "title": schema_name},
            method="json_schema",
            include_raw=True,
            strict=self._strict,
        )
        out: Any = structured.invoke([(m.role, m.content) for m in messages])
        message = out["raw"]
        content = (
            message.content if isinstance(message.content, str) else json.dumps(message.content)
        )
        metadata: dict[str, Any] = dict(message.response_metadata or {})
        usage: dict[str, Any] = dict(metadata.get("token_usage") or {})
        parsed = out.get("parsed")
        return RawResponse(
            content=content,
            parsed=parsed if isinstance(parsed, dict) else None,
            usage=usage,
            model=str(metadata.get("model_name") or model),
        )
"""Recorded model responses, so tests, CI and demos run with no API key.

Two stores share one JSONL format (one `{"key", "record"}` object per line, last
write wins):

- `ReplayStore` for structured calls, keyed by hash(model, prompt_version,
  messages, schema).
- `TrajectoryStore` for agent baselines, keyed by (system, question_id, model,
  prompt_version). An agent's messages contain DB rows whose order can vary, so
  hashing messages would miss on every re-run; the question identity is stable.
"""

from __future__ import annotations

import hashlib
import json
from pathlib import Path
from typing import Any

from ledgerline.llm.errors import ReplayMissError
from ledgerline.llm.types import ChatMessage


def canonical_json(value: Any) -> str:
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=False)


def replay_key(
    model: str, prompt_version: str, messages: list[ChatMessage], schema: dict[str, Any]
) -> str:
    payload = {
        "model": model,
        "prompt_version": prompt_version,
        "messages": [m.model_dump() for m in messages],
        "schema": schema,
    }
    return hashlib.sha256(canonical_json(payload).encode()).hexdigest()


def trajectory_key(system: str, question_id: str, model: str, prompt_version: str) -> str:
    for part in (system, question_id, model, prompt_version):
        if not part or "|" in part:
            raise ValueError(f"trajectory key part must be non-empty without '|': {part!r}")
    return "|".join((system, question_id, model, prompt_version))


class JsonlStore:
    def __init__(self, path: Path | str, what: str) -> None:
        self.path = Path(path)
        self._what = what
        self._records: dict[str, dict[str, Any]] = {}
        if self.path.exists():
            with self.path.open(encoding="utf-8") as fh:
                for line in fh:
                    if line.strip():
                        entry = json.loads(line)
                        self._records[entry["key"]] = entry["record"]

    def __contains__(self, key: str) -> bool:
        return key in self._records

    def __len__(self) -> int:
        return len(self._records)

    def get(self, key: str) -> dict[str, Any]:
        try:
            return self._records[key]
        except KeyError:
            raise ReplayMissError(key, self._what) from None

    def put(self, key: str, record: dict[str, Any]) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        with self.path.open("a", encoding="utf-8") as fh:
            fh.write(canonical_json({"key": key, "record": record}) + "\n")
        self._records[key] = record


class ReplayStore(JsonlStore):
    def __init__(self, path: Path | str) -> None:
        super().__init__(path, "call")


class TrajectoryStore(JsonlStore):
    def __init__(self, path: Path | str) -> None:
        super().__init__(path, "agent trajectory")

    def get_trajectory(
        self, system: str, question_id: str, model: str, prompt_version: str
    ) -> dict[str, Any]:
        return self.get(trajectory_key(system, question_id, model, prompt_version))

    def put_trajectory(
        self,
        system: str,
        question_id: str,
        model: str,
        prompt_version: str,
        trajectory: dict[str, Any],
    ) -> None:
        self.put(trajectory_key(system, question_id, model, prompt_version), trajectory)

def readonly_engine(url: str) -> Engine:
    """An engine that cannot write: SQLite opened `mode=ro`, Postgres read-only by default."""
    parsed = make_url(url)
    backend = parsed.get_backend_name()
    if backend == "sqlite":
        database = parsed.database
        if not database or database == ":memory:":
            raise ValueError("SQLite needs a file path to open read-only")
        path = Path(database).resolve()
        if not path.exists():
            raise ValueError(f"no SQLite file at {path}")
        uri = f"file:{quote(str(path))}?mode=ro"

        def connect() -> sqlite3.Connection:
            conn = sqlite3.connect(uri, uri=True, check_same_thread=False)
            conn.execute("PRAGMA query_only = ON")
            return conn

        return create_engine("sqlite://", creator=connect, poolclass=NullPool)
    if backend == "postgresql":
        engine = create_engine(url, pool_pre_ping=True)

        @event.listens_for(engine, "connect")
        def _readonly_default(dbapi_connection: Any, _record: Any) -> None:  # pyright: ignore[reportUnusedFunction]
            with dbapi_connection.cursor() as cursor:
                cursor.execute("SET SESSION CHARACTERISTICS AS TRANSACTION READ ONLY")
            dbapi_connection.commit()

        return engine
    raise ValueError(f"unsupported database backend {backend!r}")


def run(
    engine: Engine,
    stmt: Executable,
    *,
    timeout_s: float = DEFAULT_TIMEOUT_S,
    row_cap: int = DEFAULT_ROW_CAP,
) -> QueryResult:
    """Run one statement in a read-only transaction, then roll back."""
    dialect = engine.dialect.name
    with engine.connect() as conn:
        raw: Any = conn.connection.dbapi_connection
        try:
            if dialect == "postgresql":
                conn.exec_driver_sql("SET TRANSACTION READ ONLY")
                conn.exec_driver_sql(f"SET LOCAL statement_timeout = {int(timeout_s * 1000)}")
            elif dialect == "sqlite":
                deadline = time.monotonic() + timeout_s
                raw.set_progress_handler(lambda: int(time.monotonic() > deadline), 1000)
            result = conn.execute(stmt)
            columns = list(result.keys())
            rows = [tuple(row) for row in result.fetchmany(row_cap + 1)]
        except DBAPIError as exc:
            if _is_timeout(exc):
                raise QueryTimeoutError(f"query cancelled after {timeout_s}s") from exc
            raise ExecutionError(str(exc.orig)) from exc
        finally:
            if dialect == "sqlite":
                raw.set_progress_handler(None, 0)
            conn.rollback()
    return QueryResult(columns, rows[:row_cap], len(rows) > row_cap)


def _is_timeout(exc: DBAPIError) -> bool:
    orig = exc.orig
    if type(orig).__name__ == "QueryCanceled":
        return True
    return isinstance(orig, sqlite3.OperationalError) and "interrupted" in str(orig)


class Executor:
    def __init__(
        self,
        engine: Engine,
        *,
        audit: AuditLog | None = None,
        timeout_s: float = DEFAULT_TIMEOUT_S,
        row_cap: int = DEFAULT_ROW_CAP,
    ) -> None:
        self.engine = engine
        self.audit = audit
        self.timeout_s = timeout_s
        self.row_cap = row_cap

    def execute(
        self,
        plan: QueryPlan,
        *,
        question: str = "",
        intent_json: str = "",
        validation_json: str = "",
        schema_id: str | None = None,
        cost_usd: float | None = None,
        model: str | None = None,
        trace_id: str | None = None,
    ) -> Execution:
        trace = trace_id or str(uuid.uuid4())
        stmt = build(plan)
        sql, params = compile_sql(stmt, self.engine.dialect.name)
        entry = AuditEntry(
            trace_id=trace,
            schema_id=schema_id,
            question=question,
            intent=intent_json,
            validation=validation_json,
            sql=sql,
            params_hash=params_hash(params),
            row_count=None,
            cost_usd=cost_usd,
            model=model,
            outcome="error",
        )
        try:
            result = run(self.engine, stmt, timeout_s=self.timeout_s, row_cap=self.row_cap)
        except ExecutionError:
            if self.audit is not None:
                self.audit.record(entry)

# Ledgerline v0.1 — plan (rev 2, after Fable 5.2 Bounce)

**Outcome:** A Python service and CLI where a natural-language question about a registered schema becomes a validated structured intent, and then a parameterized SQL query run on a read-only connection. A pre-registered eval reports coverage, accuracy, false-refusal rate and wrong-answer rate per schema. It covers 6 schemas (the synthetic fieldservice schema plus 5 Spider databases) and compares against two baselines: a one-shot LLM that writes SQL, and LangChain's SQL agent. Everything except live evals runs with no API key.

**PRs:** three PRs, stacked, each merged with a merge commit.
- **PR1:** P1 + P2, foundations and the correctness core. Main works on its own: a CLI that loads a registry and validates and compiles intents. No LLM is involved.
- **PR2:** P3 + P4, the pipeline and the eval. It needs PR1's core.
- **PR3:** P5, the UI and docs.

The security-relevant core is reviewed apart from routine work.

## Stack (fixed)
- **Core:** Python 3.12 (uv), FastAPI, uvicorn, Pydantic v2, pydantic-settings, PyYAML, SQLAlchemy 2 Core, psycopg, sqlite3.
- **LLM:** LangGraph and langchain-openai (OpenRouter). langchain-community for the SQL agent baseline.
- **CLI and dev:** Typer, httpx, Faker.
- **Tests and checks:** pytest, pytest-asyncio, pytest-cov, hypothesis, ruff, pyright (strict on `ledgerline/core/`), detect-secrets.
- **Web:** Vite + TypeScript, a static build served by FastAPI. Playwright for one smoke test.
- **Infra:** Docker Compose (Postgres 16, api).
- New dependencies outside this list need a callback.

## Definitions (fixed before any code)
- **Reference time:** the config value `LEDGERLINE_NOW` (default `2026-06-30T00:00:00Z`). It's used by the seeder, by the prompt's reference date, and by relative-date resolution. Gold results are computed against it.
- **Gold categories:**
  - `in_schema` is answerable from registry-visible data, and its `gold_sql` must execute and return a non-empty, deterministic result.
  - `out_of_schema` asks for data that does not exist in the DB, such as competitor prices. The gold answer is a refusal.
  - `should_refuse` asks for something that exists but is deliberately excluded or unsafe: a hidden field, a write, a request to delete. The gold answer is a refusal.
  - Each item carries `difficulty` (easy|medium|hard) and `features` (join, group, having, date, distinct, order_limit).
- **Metrics, per schema × system × model:**
  - coverage: share of in_schema items the system attempts
  - execution accuracy on in_schema
  - accuracy given attempted
  - false-refusal rate on in_schema
  - wrong-answer rate given answered, which is the governance metric
  - refusal recall on refusal items
  - execution-error rate
  - mean cost and p50 latency
  - Wilson 95% intervals on every rate
- **Result comparison:**
  - project to the gold column count
  - compare rows as multisets, or as sequences when the gold has ORDER BY
  - cast Decimal to float with tolerance 1e-6
  - NULL is never equal to 0
  - normalize SQLite date strings to ISO
  - compare gold under the same row cap
  - extra columns count as a mismatch unless the gold columns are a subset by value
- **Baselines:**
  - **oneshot:** schema DDL plus the registry descriptions go into the prompt, and the LLM returns SQL that is executed read-only.
  - **sql_agent:** LangChain's SQL agent with the same descriptions in its prompt. It is scored on the rows of its last successful `sql_db_query` call. A refusal is "no query executed" or a final answer matching a fixed refusal pattern list.

  All three systems run at temperature 0. Fieldservice runs 3 trials and reports the mean. Spider runs one trial.

## Phases

### P1 — Foundations, gold, pre-registration (PR1)
1. Scaffold:
   - uv project, Typer CLI, ruff and pyright config, pytest config with **zero skips allowed** (verify fails on any skip)
   - `verify.sh` and a GitHub Actions workflow running it (Postgres service), `.env.example`, Apache-2.0
2. **Compose:** Postgres with roles `ledgerline_owner` and `ledgerline_ro` (SELECT only). One seed path, `uv run ledgerline seed`, which compose calls.
   - **Fieldservice schema:** customers, sites, technicians, work_orders, parts, work_order_parts, invoices, plus a `users` table with `password_hash` and `email` to exercise hidden fields.
   - Faker with a fixed seed, about 5k work orders over 24 months ending at `LEDGERLINE_NOW`.
3. **Model gateway** (`ledgerline/llm/`):
   - An OpenRouter `ChatOpenAI` with structured output (json_schema), pinned to providers that support it, with a Pydantic parse fallback. Unparseable output is a typed `ParseError`.
   - **Replay:** a JSONL store keyed by hash(model, prompt_version, messages, schema), with `record|replay|live` modes. A miss raises.
   - **Agent trajectory replay** is keyed by (system, question_id, model, prompt_version), not by message hash.
   - **Cost:** taken from OpenRouter's usage cost, or estimated from the models-endpoint pricing.
   - **Budget guard:** a worst-case pre-check (max_tokens × price) before each call, a hard stop at `--budget`, 429 backoff, and a checkpointed results JSONL so runs resume and produce a partial report.
   - **Backstop:** `.env.example` tells Cristian to set a credit limit on the OpenRouter key.
4. **Model spike** (live, budget $0.50): pick a default cheap model and a second model that honour json_schema on OpenRouter. Record the choice in ADR-004.
5. **Gold set** `evals/gold/fieldservice.jsonl`:
   - ≥60 items: ≥35 in_schema with a difficulty mix of ≥10 hard, ≥12 out_of_schema, ≥10 should_refuse.
   - A test asserts every in_schema `gold_sql` executes and returns a non-empty, deterministic result (run twice, compared).
6. **Spider pre-registration:**
   - `scripts/fetch_benchmarks.py` downloads Spider dev with a pinned checksum and licence notice into gitignored `data/`.
   - `evals/gold/spider_selection.py` uses a fixed seed and stratifies by Spider hardness across 5 databases picked by seed. It commits only **question IDs** (`spider_ids.json`), never question text.
7. **Gold manifest:** `evals/gold/MANIFEST.json` holds the gold file hashes, and a test checks them. Changing gold after P1 requires `evals/gold/CHANGELOG.md` with a reason.
8. **ADR skeletons with predictions:** intent-not-SQL, LangGraph boundary, fan-out rule, replay and budget, benchmark methodology. Predictions only, committed now.

**Exit:** `./verify.sh` is green. The gold-executes test, budget-guard tests (pre-check, abort, resume), replay-miss test and manifest test all pass. The spike is recorded.

### P2 — Registry, validator, SQL builder, executor (PR1, plain Python, `ledgerline/core/`)
1. **Registry model** (`schema_version: 1`, `draft: bool`):
   - **entities** are DB tables or DB views only. There is no SQL in the YAML.
   - **fields:** name, real identifier, type, description, aliases, filterable, aggregatable, hidden, categorical.
   - **relationships**, each with a cardinality (many_to_one) and a `source: declared|inferred`.
   - **allowed_values** for categorical fields only.

   Registry-safe slugs map to real identifiers, so spaces, dots and case are handled.
2. **`ledgerline registry init <db-url> -o file.yaml [--values]`:**
   - **Introspection:** uses the SQLAlchemy inspector. Infers relationships from `<x>_id` → a primary key when no foreign key is declared, marked `inferred`.
   - **Auto-hide:** hides fields matching `password|hash|token|secret|ssn|email|phone`.
   - **Categoricals:** a field is categorical when it has ≤50 distinct values, isn't hidden and isn't name-like.
   - Marks the file `draft: true`, and warns if the role has write privileges.
   - Running it twice on the same DB gives byte-identical output.
3. **Intent model:**
   - **intent_type:** summary | grouped_summary | row_list | unanswerable.
   - **entity** (the base).
   - **select** (for row_list).
   - **metrics:** count | count_distinct | sum | avg | min | max, with distinct.
   - **filters:** eq | neq | gt | gte | lt | lte | between | in | is_null | ilike_contains, grouped as any_of / all_of, one level deep.
   - **having** (on metrics).
   - **group_by:** fields plus calendar grain day | iso_week | month | quarter | year.
   - **date_range:** relative tokens (last_n_days, last_month, last_quarter, ytd, …) or absolute, applied to one named date field.
   - **order_by**, **limit**, **unanswerable_reason**.

   Code resolves relative dates against `LEDGERLINE_NOW`, never the LLM.
4. **Validator**, with typed errors for:
   - unknown entity or field
   - a hidden field
   - a field that is not filterable or not aggregatable
   - aggregate vs type, operator vs type, value type vs field type
   - a value not in allowed_values (case-insensitive, alias-resolved) for categoricals
   - limit ≤0 or above the cap
   - an empty `in`
   - a reversed `between`
   - date_range on a non-date field, or a date_range that duplicates a filter on the same field
   - intent_type vs populated fields (summary needs metrics, row_list needs select)
   - having without a group_by
   - an unreachable field (no declared or inferred path, more than 2 hops)
   - **an ambiguous path** (more than one shortest path)
   - **fan_out:** a join from the base entity along a one-to-many edge while any metric is aggregated
   - a draft registry without `allow_draft`
5. **SQL builder:** SQLAlchemy Core only. Identifiers come from the registry. Values are bound parameters. Joins run along many-to-one edges only. Calendar grains compile per dialect (`date_trunc` on Postgres, `strftime` on SQLite, ISO week on both).
6. **Executor:**
   - On every connection, Postgres runs `BEGIN READ ONLY` with `statement_timeout`. SQLite opens `mode=ro` with a progress-handler interrupt as its timeout.
   - Row cap.
   - Writes go to an audit table `ask_log` in the app DB (trace id, question, intent, validation, sql, params hash, row count, cost, model). Data rows are never logged.

**Exit:**
- one named test per validator rule (each listed one)
- a Hypothesis property using sentinel values: every filter value appears in params and never in the SQL text
- injection fixtures leave the DB unchanged
- a **differential oracle**: 30 hand-written intents, each with expected rows on the fieldservice fixture DB, on both engines
- cross-dialect grain tests
- write attempts fail on both engines, and the timeout fires on both
- `registry init` is deterministic and auto-hides fields
- **branch** coverage ≥95% per module on `validate.py` and `sqlbuild.py`, and ≥85% on `executor.py` and `registry_init.py`, checked per module by `scripts/check_coverage.py`

### P3 — Pipeline, API, CLI (PR2)
1. **LangGraph graph:** interpret → validate → (one retry with the typed errors) → execute → present.
   - **present is deterministic** (columns, rows, a formatted SQL string). There is no LLM call, and no rows are sent to OpenRouter.
   - **Interpret prompt:** a compact registry view in a fixed order. When it's over budget, truncation is deterministic: allowed values drop beyond 20 per field, then descriptions shorten. The truncation is logged.
   - Hidden fields are never in the prompt.
   - `unanswerable`, or a second validation failure, gives a refusal with a reason and `sql: null`.
2. **`config/schemas.yaml`:** schema_id → registry path + DB URL env var.
3. **FastAPI** `POST /ask` returns `{intent, validation, sql, params, rows, columns, refusal, trace_id, cost_usd}`. Also `GET /schemas`, `GET /schemas/{id}/examples` (canned questions available in replay), `GET /healthz`.
4. **CLI:** `ledgerline ask`.

**Exit:** replay API tests cover answerable, out-of-registry refusal, hidden-field refusal, retry-then-success, and **init → ask on a Spider DB**. The `ask_log` row is written.

class _Validator:
    def __init__(self, intent: Intent, registry: Registry, now: datetime, row_cap: int) -> None:
        self.intent = intent
        self.registry = registry
        self.now = now
        self.row_cap = row_cap
        self.issues: list[Issue] = []
        self.aggregated = bool(intent.metrics)
        self.base = ""
        self.base_def: EntityDef | None = None
        self.paths: dict[str, _PathInfo] = {}
        self.join_aliases: dict[tuple[tuple[int, bool], ...], str] = {}
        self.joins: list[JoinStep] = []

    def fail(self, code: ErrorCode, location: str, message: str) -> None:
        self.issues.append(Issue(code, location, message))

    # entity and field lookup

    def _find_entity(self, name: str) -> str | None:
        lowered = name.lower()
        if lowered in self.registry.entities:
            return lowered
        for slug, entity in self.registry.entities.items():
            if lowered in (alias.lower() for alias in entity.aliases):
                return slug
        return None

    @staticmethod
    def _find_field(entity: EntityDef, name: str) -> str | None:
        lowered = name.lower()
        if lowered in entity.fields:
            return lowered
        for slug, fdef in entity.fields.items():
            if lowered in (alias.lower() for alias in fdef.aliases):
                return slug
        return None

    def _alias_for(self, path: tuple[_Edge, ...]) -> str:
        if not path:
            return "t0"
        key = tuple((edge.rel, edge.down) for edge in path)
        if key in self.join_aliases:
            return self.join_aliases[key]
        parent = self._alias_for(path[:-1])
        alias = f"t{len(self.join_aliases) + 1}"
        self.join_aliases[key] = alias
        edge = path[-1]
        src = self.registry.entities[edge.src]
        dst = self.registry.entities[edge.dst]
        self.joins.append(
            JoinStep(
                alias=alias,
                parent_alias=parent,
                parent_column=src.fields[edge.src_field].column,
                table=dst.table,
                db_schema=dst.db_schema,
                column=dst.fields[edge.dst_field].column,
            )
        )
        return alias

    def resolve(self, ref: str, location: str) -> tuple[ResolvedField, FieldDef] | None:
        assert self.base_def is not None
        if "." in ref:
            entity_name, field_name = ref.split(".", 1)
            entity_slug = self._find_entity(entity_name)
            if entity_slug is None:
                self.fail("unknown_entity", location, f"no entity {entity_name!r}")
                return None
        else:
            entity_slug, field_name = self.base, ref
        entity = self.registry.entities[entity_slug]
        field_slug = self._find_field(entity, field_name)
        if field_slug is None:
            self.fail("unknown_field", location, f"no field {field_name!r} on {entity_slug}")
            return None
        fdef = entity.fields[field_slug]
        if fdef.hidden:
            self.fail("hidden_field", location, f"{entity_slug}.{field_slug} is hidden")
            return None
        info = self.paths.get(entity_slug)
        if info is None:
            self.fail(
                "unreachable_field",
                location,
                f"{entity_slug} has no path from {self.base} within {MAX_HOPS} hops",
            )
            return None
        if info.count > 1:
            self.fail(
                "ambiguous_path",
                location,
                f"{info.count} equally short paths from {self.base} to {entity_slug}",
            )
            return None
        if any(edge.down for edge in info.path):
            if self.aggregated:
                self.fail(
                    "fan_out",
                    location,
                    f"joining {self.base} to {entity_slug} crosses a one-to-many edge "
                    "while aggregating, which would double count",
                )
            else:
                self.fail(
                    "one_to_many_join",
                    location,
        }
    )


def _invoice(
    rng: random.Random, iid: int, woid: int, issued: date, amount: Decimal, today: date
) -> Row:
    due = issued + timedelta(days=30)
    paid_at: date | None = None
    roll = rng.random()
    if roll < 0.02:
        status = "void"
    else:
        paid_at = issued + timedelta(days=rng.randint(0, 50))
        if roll < 0.88 and paid_at < today:
            status = "paid"
        else:
            paid_at = None
            status = "overdue" if due < today else "unpaid"
    return {
        "id": iid,
        "work_order_id": woid,
        "issued_at": issued,
        "due_at": due,
        "amount": amount,
        "status": status,
        "paid_at": paid_at,
    }


def _chunks(rows: Sequence[Row], size: int) -> Iterable[Sequence[Row]]:
    for i in range(0, len(rows), size):
        yield rows[i : i + size]


def load(engine: Engine, dataset: Dataset, ro_role: str | None = "ledgerline_ro") -> None:
    """Drop and recreate the fieldservice tables, then insert every row.

    On Postgres, SELECT is granted to `ro_role` when that role exists.
    """
    schema.metadata.drop_all(engine)
    schema.metadata.create_all(engine)
    with engine.begin() as conn:
        for table in schema.TABLES_IN_LOAD_ORDER:
            rows = dataset.tables[table.name]
            for chunk in _chunks(rows, 2000):
                conn.execute(table.insert(), list(chunk))
        if engine.dialect.name == "postgresql" and ro_role is not None:
            exists = conn.execute(
                text("SELECT 1 FROM pg_roles WHERE rolname = :r"), {"r": ro_role}
            ).first()
            if exists:
                _grant_readonly(conn, ro_role)


def _grant_readonly(conn: Connection, ro_role: str) -> None:
    """SELECT on every table, except column-level SELECT on tables with denied columns.

    A table-level grant would cover every column, so those tables get the table grant
    revoked and a column list granted instead."""
    for table in schema.TABLES_IN_LOAD_ORDER:
        denied = schema.RO_DENIED_COLUMNS.get(table.name, frozenset())
        conn.execute(text(f'REVOKE ALL ON "{table.name}" FROM "{ro_role}"'))
        if not denied:
            conn.execute(text(f'GRANT SELECT ON "{table.name}" TO "{ro_role}"'))
            continue
        allowed = ", ".join(f'"{c.name}"' for c in table.columns if c.name not in denied)
        conn.execute(text(f'GRANT SELECT ({allowed}) ON "{table.name}" TO "{ro_role}"'))

"""Row comparison for tests: Decimal/float within tolerance, dates as ISO, NULL is NULL."""

from __future__ import annotations

import math
from collections.abc import Sequence
from datetime import date, datetime
from decimal import Decimal
from typing import Any

Cell = float | str | bool | int | None


def _cell(value: Any) -> Cell:
    if isinstance(value, bool) or value is None:
        return value
    if isinstance(value, Decimal | float | int):
        return float(value)
    if isinstance(value, datetime):
        return value.isoformat(sep=" ")
    if isinstance(value, date):
        return value.isoformat()
    return str(value)


def normalise(rows: Sequence[Sequence[Any]], ordered: bool) -> list[tuple[Cell, ...]]:
    out = [tuple(_cell(v) for v in row) for row in rows]
    if ordered:
        return out

    def key(row: tuple[Cell, ...]) -> tuple[str, ...]:
        return tuple(f"{v:.6f}" if isinstance(v, float) else repr(v) for v in row)

    return sorted(out, key=key)


def same_rows(a: Sequence[Sequence[Any]], b: Sequence[Sequence[Any]], ordered: bool) -> bool:
    left, right = normalise(a, ordered), normalise(b, ordered)
    if len(left) != len(right):
        return False
    for row_a, row_b in zip(left, right, strict=True):
        if len(row_a) != len(row_b):
            return False
        for x, y in zip(row_a, row_b, strict=True):
            if isinstance(x, float) and isinstance(y, float):
                if not math.isclose(x, y, rel_tol=1e-9, abs_tol=1e-6):
                    return False
            elif x != y:
                return False
    return True
